```python
from __future__ import annotations

import os
from collections.abc import Mapping
from dataclasses import dataclass
from enum import StrEnum
from pathlib import Path

from platformdirs import user_data_dir

APP_NAME = "mlops-agents"
ENV_PREFIX = "MLOPS_AGENTS_"
DEFAULT_PORT = 8765


class RunMode(StrEnum):
    LOCAL = "local"
    SERVER = "server"


class ConfigError(Exception):
    """Startup configuration is invalid. The message is shown to the user as is."""


@dataclass(frozen=True)
class Settings:
    mode: RunMode
    data_dir: Path
    host: str
    port: int
    password: str | None
    master_key: str | None
    access_token: str | None
    allowed_hosts: tuple[str, ...]
    dev_origins: tuple[str, ...]
    trusted_proxies: tuple[str, ...]


def _get(env: Mapping[str, str], name: str) -> str | None:
    value = env.get(ENV_PREFIX + name, "").strip()
    return value or None


def _csv(value: str | None) -> tuple[str, ...]:
    if value is None:
        return ()
    return tuple(part.strip() for part in value.split(",") if part.strip())


def default_data_dir() -> Path:
    # Non-roaming on Windows: SQLite must stay on local disk.
    return Path(user_data_dir(APP_NAME, appauthor=False, roaming=False))


def load_settings(
    mode: RunMode, port: int = DEFAULT_PORT, env: Mapping[str, str] | None = None
) -> Settings:
    source = os.environ if env is None else env
    home = _get(source, "HOME")
    data_dir = Path(home) if home else default_data_dir()
    allowed_hosts = _csv(_get(source, "ALLOWED_HOSTS"))
    password: str | None = None
    master_key: str | None = None
    if mode is RunMode.SERVER:
        password = _get(source, "PASSWORD")
        master_key = _get(source, "MASTER_KEY")
        missing = [
            name
            for name, value in (
                ("MLOPS_AGENTS_PASSWORD", password),
                ("MLOPS_AGENTS_MASTER_KEY", master_key),
                ("MLOPS_AGENTS_ALLOWED_HOSTS", ",".join(allowed_hosts)),
            )
            if not value
        ]
        if missing:
            raise ConfigError(
                "Server mode needs these environment variables: "
                + ", ".join(missing)
                + ". Set them and start again."
            )
    return Settings(
        mode=mode,
        data_dir=data_dir,
        host="0.0.0.0" if mode is RunMode.SERVER else "127.0.0.1",  # noqa: S104 -- server mode sits behind a reverse proxy
        port=port,
        password=password,
        master_key=master_key,
        access_token=_get(source, "ACCESS_TOKEN"),
        allowed_hosts=allowed_hosts,
        dev_origins=_csv(_get(source, "DEV_ORIGINS")),
        trusted_proxies=_csv(_get(source, "TRUSTED_PROXIES")),
    )
```

`MLOPS_AGENTS_ACCESS_TOKEN` (optional) fixes the local access token instead of generating one; scripts and the end-to-end test use it. `MLOPS_AGENTS_DEV_ORIGINS` lists extra allowed origins, e.g. `http://localhost:5173` for the Vite dev server.

`backend/src/mlops_agents/paths.py`:

```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True)
class DataPaths:
    root: Path

    @property
    def app_db(self) -> Path:
        return self.root / "app.db"

    @property
    def keys_file(self) -> Path:
        return self.root / "keys.enc"

    @property
    def cookie_secret(self) -> Path:
        return self.root / "cookie_secret"

    @property
    def backups(self) -> Path:
        return self.root / "backups"

    @property
    def projects(self) -> Path:
        return self.root / "projects"

    @property
    def bin(self) -> Path:
        return self.root / "bin"

    def project_dir(self, project_id: str) -> Path:
        return self.projects / project_id

    def project_db(self, project_id: str) -> Path:
        return self.project_dir(project_id) / "project.db"

    def repo_dir(self, project_id: str) -> Path:
        return self.project_dir(project_id) / "repo"

    def ensure(self) -> None:
        for folder in (self.root, self.backups, self.projects, self.bin):
            folder.mkdir(parents=True, exist_ok=True)
```

```python
import sqlite3
from pathlib import Path

import pytest

from mlops_agents.storage.migrate import MigrationError, migrate


def _write(folder: Path, name: str, sql: str) -> None:
    folder.mkdir(parents=True, exist_ok=True)
    (folder / name).write_text(sql, encoding="utf-8")
```

```python
from __future__ import annotations

import sqlite3
from pathlib import Path
from typing import Any

import aiosqlite
from sqlalchemy import event
from sqlalchemy.ext.asyncio import AsyncEngine, create_async_engine

PRAGMAS: tuple[str, ...] = (
    "PRAGMA journal_mode=WAL",
    "PRAGMA busy_timeout=5000",
    "PRAGMA foreign_keys=ON",
    "PRAGMA synchronous=NORMAL",
)


def create_engine_for(path: Path) -> AsyncEngine:
    engine = create_async_engine(f"sqlite+aiosqlite:///{path.as_posix()}")

    @event.listens_for(engine.sync_engine, "connect")
    def on_connect(dbapi_connection: Any, connection_record: Any) -> None:
        del connection_record
        cursor = dbapi_connection.cursor()
        for pragma in PRAGMAS:
            cursor.execute(pragma)
        cursor.close()

    return engine


def connect_sync(path: Path) -> sqlite3.Connection:
    """A stdlib connection in autocommit mode: callers manage transactions explicitly."""
    conn = sqlite3.connect(path, isolation_level=None)
    for pragma in PRAGMAS:
        conn.execute(pragma)
    return conn


async def connect_checkpointer(path: Path) -> aiosqlite.Connection:
    conn = await aiosqlite.connect(path)
    for pragma in PRAGMAS:
        await conn.execute(pragma)
    return conn
```

```python
from __future__ import annotations

import shutil
from datetime import UTC, datetime
from pathlib import Path

from mlops_agents.paths import DataPaths
from mlops_agents.storage.sqlite import connect_sync


def _stamp() -> str:
    return datetime.now(UTC).strftime("%Y%m%dT%H%M%S%fZ")


def _vacuum_into(db_path: Path, dest: Path) -> None:
    dest.parent.mkdir(parents=True, exist_ok=True)
    conn = connect_sync(db_path)
    try:
        conn.execute("VACUUM INTO ?", (str(dest),))
    finally:
        conn.close()


def _prune(entries: list[Path], keep: int) -> None:
    if len(entries) <= keep:
        return
    for old in sorted(entries)[:-keep]:
        if old.is_dir():
            shutil.rmtree(old)
        else:
            old.unlink()


def backup_db(db_path: Path, backup_dir: Path, keep: int = 10) -> Path:
    """A consistent copy of a database, safe while the app is writing to it."""
    dest = backup_dir / f"{_stamp()}.db"
    _vacuum_into(db_path, dest)
    _prune(list(backup_dir.glob("*.db")), keep)
    return dest


def snapshot_all(paths: DataPaths, keep: int = 10) -> Path:
    snapshots = paths.backups / "snapshots"
    dest = snapshots / _stamp()
    dest.mkdir(parents=True, exist_ok=True)
    if paths.app_db.exists():
        _vacuum_into(paths.app_db, dest / "app.db")
    for project_db in sorted(paths.projects.glob("*/project.db")):
        _vacuum_into(project_db, dest / "projects" / f"{project_db.parent.name}.db")
    _prune([p for p in snapshots.iterdir() if p.is_dir()], keep)
    return dest
```

`backend/src/mlops_agents/storage/migrate.py`:

```python
from __future__ import annotations

import re
import sqlite3
from dataclasses import dataclass
from importlib.resources import files
from importlib.resources.abc import Traversable
from pathlib import Path
from typing import Literal

from mlops_agents.clock import now_iso
from mlops_agents.storage.backup import backup_db
from mlops_agents.storage.sqlite import connect_sync

DbKind = Literal["app", "project"]
_NAME = re.compile(r"^(\d{4})_[a-z0-9_]+\.sql$")


class MigrationError(Exception):
    pass


@dataclass(frozen=True)
class Migration:
    version: int
    name: str
    sql: str


def load_migrations(kind: DbKind, source: Path | None = None) -> list[Migration]:
    folder: Traversable = (
        source if source is not None else files("mlops_agents.storage").joinpath("migrations", kind)
    )
    found: list[Migration] = []
    for entry in folder.iterdir():
        match = _NAME.match(entry.name)
        if match:
            found.append(Migration(int(match.group(1)), entry.name, entry.read_text("utf-8")))
    found.sort(key=lambda m: m.version)
    for expected, migration in enumerate(found, start=1):
        if migration.version != expected:
            raise MigrationError(
                f"{kind} migrations must be numbered 0001, 0002, ... without gaps; "
                f"found {migration.name}"
            )
    return found


def _current_version(conn: sqlite3.Connection) -> int:
    conn.execute(
        "CREATE TABLE IF NOT EXISTS schema_version "
        "(version INTEGER NOT NULL, applied_at TEXT NOT NULL)"
    )
    row = conn.execute("SELECT MAX(version) FROM schema_version").fetchone()
    return int(row[0]) if row is not None and row[0] is not None else 0


def migrate(
    db_path: Path, kind: DbKind, backup_dir: Path, *, source: Path | None = None
) -> list[int]:
    """Apply pending migrations, each in its own transaction. Returns the versions applied."""
    migrations = load_migrations(kind, source)
    db_path.parent.mkdir(parents=True, exist_ok=True)
    conn = connect_sync(db_path)
    try:
        current = _current_version(conn)
        latest = migrations[-1].version if migrations else 0
        if current > latest:
            raise MigrationError(
                f"{db_path.name} was created by a newer version of the app "
                f"(schema {current}, this app knows {latest}). Update the app."
            )
        pending = [m for m in migrations if m.version > current]
        if not pending:
            return []
        if current > 0:
            backup_db(db_path, backup_dir)
        for migration in pending:
            try:
                # executescript leaves the transaction opened by BEGIN open on success.
                conn.executescript("BEGIN;\n" + migration.sql)
                conn.execute(
                    "INSERT INTO schema_version (version, applied_at) VALUES (?, ?)",
                    (migration.version, now_iso()),
                )
                conn.execute("COMMIT")
            except sqlite3.Error as exc:
                if conn.in_transaction:
                    conn.execute("ROLLBACK")
                raise MigrationError(f"Migration {migration.name} failed: {exc}") from exc
        return [m.version for m in pending]
    finally:
        conn.close()
```

```python
from __future__ import annotations

import asyncio

from sqlalchemy.ext.asyncio import AsyncEngine

from mlops_agents.paths import DataPaths
from mlops_agents.storage.migrate import migrate
from mlops_agents.storage.sqlite import create_engine_for


class Databases:
    """One migrated engine for app.db and one per project.db, created on first use."""

    def __init__(self, paths: DataPaths) -> None:
        self._paths = paths
        self._app: AsyncEngine | None = None
        self._projects: dict[str, AsyncEngine] = {}
        self._lock = asyncio.Lock()

    async def app(self) -> AsyncEngine:
        async with self._lock:
            if self._app is None:
                await asyncio.to_thread(
                    migrate, self._paths.app_db, "app", self._paths.backups / "migrations" / "app"
                )
                self._app = create_engine_for(self._paths.app_db)
            return self._app

    async def project(self, project_id: str) -> AsyncEngine:
        async with self._lock:
            engine = self._projects.get(project_id)
            if engine is None:
                db_path = self._paths.project_db(project_id)
                backup_dir = self._paths.backups / "migrations" / "projects" / project_id
                await asyncio.to_thread(migrate, db_path, "project", backup_dir)
                engine = create_engine_for(db_path)
                self._projects[project_id] = engine
            return engine

    async def close_project(self, project_id: str) -> None:
        """Release the project's database files (needed before deleting them on Windows)."""
        async with self._lock:
            engine = self._projects.pop(project_id, None)
        if engine is not None:
            await engine.dispose()

    async def dispose(self) -> None:
        async with self._lock:
            engines = list(self._projects.values())
            if self._app is not None:
                engines.append(self._app)
            self._projects.clear()
            self._app = None
        for engine in engines:
            await engine.dispose()
```

```python
import keyring
from keyring.errors import KeyringError, PasswordDeleteError

from mlops_agents.config import APP_NAME
from mlops_agents.credentials.keystore import KeyStoreError

SERVICE = APP_NAME


class KeyringBackendLike(Protocol):
    def get_password(self, service: str, username: str) -> str | None: ...

    def set_password(self, service: str, username: str, password: str) -> None: ...

    def delete_password(self, service: str, username: str) -> None: ...


class KeyringStore:
    """Keys in the OS keyring (Windows Credential Manager, macOS Keychain, Secret Service)."""

    def __init__(self, backend: KeyringBackendLike | None = None) -> None:
        self._backend: KeyringBackendLike = (
            backend if backend is not None else keyring.get_keyring()
        )

    def get(self, ref: str) -> str | None:
        try:
            return self._backend.get_password(SERVICE, ref)
        except KeyringError as exc:
            raise KeyStoreError(
                f"The system keyring could not be read ({type(exc).__name__})."
            ) from None

    def set(self, ref: str, value: str) -> None:
        try:
            self._backend.set_password(SERVICE, ref, value)
            stored = self._backend.get_password(SERVICE, ref)
        except KeyringError as exc:
            raise KeyStoreError(
                f"The system keyring refused to store the key ({type(exc).__name__})."
            ) from None
        if stored != value:
            raise KeyStoreError(
                "The system keyring did not keep the key. Check that a keyring service is running."
            )

    def delete(self, ref: str) -> None:
        try:
            self._backend.delete_password(SERVICE, ref)
        except PasswordDeleteError:
            return
        except KeyringError as exc:
            raise KeyStoreError(
                f"The system keyring could not delete the key ({type(exc).__name__})."
```

```python
from __future__ import annotations

import base64
import json
import os
import threading
from pathlib import Path

from cryptography.fernet import Fernet, InvalidToken
from cryptography.hazmat.primitives.kdf.scrypt import Scrypt

from mlops_agents.credentials.keystore import KeyStoreError

MAGIC = b"MLAK1"
SALT_LEN = 16


class EncryptedFileStore:
    """Keys in one encrypted JSON document, for server mode where no OS keyring exists.

    File layout: MAGIC, a random scrypt salt, then a Fernet token. The Fernet key is derived
    from MLOPS_AGENTS_MASTER_KEY with scrypt. Writes replace the file atomically.
    """

    def __init__(self, path: Path, master_key: str) -> None:
        self._path = path
        self._master_key = master_key
        self._lock = threading.Lock()
        self._salt = b""
        self._fernet: Fernet | None = None
        self._data: dict[str, str] = {}

    def _derive(self, salt: bytes) -> Fernet:
        kdf = Scrypt(salt=salt, length=32, n=2**15, r=8, p=1)
        return Fernet(base64.urlsafe_b64encode(kdf.derive(self._master_key.encode("utf-8"))))

    def _load(self) -> Fernet:
        if self._fernet is not None:
            return self._fernet
        if not self._path.exists():
            self._salt = os.urandom(SALT_LEN)
            self._fernet = self._derive(self._salt)
            self._data = {}
            return self._fernet
        raw = self._path.read_bytes()
        header = len(MAGIC) + SALT_LEN
        if not raw.startswith(MAGIC) or len(raw) <= header:
            raise KeyStoreError(f"{self._path.name} is not a key file written by this app.")
        salt = raw[len(MAGIC) : header]
        fernet = self._derive(salt)
        try:
            payload = fernet.decrypt(raw[header:])
        except InvalidToken:
            raise KeyStoreError(
                "MLOPS_AGENTS_MASTER_KEY does not unlock keys.enc. Start with the master key that "
                "was used when the keys were saved, or delete keys.enc and add your keys again."
            ) from None
        loaded: object = json.loads(payload)
        if not isinstance(loaded, dict):
            raise KeyStoreError(f"{self._path.name} is damaged.")
        self._data = {str(k): str(v) for k, v in loaded.items()}  # pyright: ignore[reportUnknownVariableType] -- json.loads returns untyped data
        self._salt = salt
        self._fernet = fernet
        return fernet

    def _save(self, fernet: Fernet) -> None:
        token = fernet.encrypt(json.dumps(self._data).encode("utf-8"))
        tmp = self._path.with_name(self._path.name + ".tmp")
        tmp.write_bytes(MAGIC + self._salt + token)
        tmp.replace(self._path)

    def check(self) -> None:
        with self._lock:
            self._load()

    def get(self, ref: str) -> str | None:
        with self._lock:
            self._load()
            return self._data.get(ref)

    def set(self, ref: str, value: str) -> None:
        with self._lock:
            fernet = self._load()
            self._data[ref] = value
            self._save(fernet)

    def delete(self, ref: str) -> None:
        with self._lock:
            fernet = self._load()
            if self._data.pop(ref, None) is not None:
                self._save(fernet)
```

```python
"""Keeping secrets out of URLs and text: base URL validation, URL and value scrubbing."""

from __future__ import annotations

import base64
import re
import unicodedata
from collections.abc import Iterable
from urllib.parse import SplitResult, quote, quote_plus, unquote_plus, urlsplit

MASK = "***"

# Parameter names that carry a credential, compared after _normalized_name. Credentials belong
# in the key store, never in a base URL; this list is defense in depth and cannot be complete.
_CREDENTIAL_PARAMS = frozenset(
    {
        "api_key", "apikey", "api_token", "x_api_key", "x_goog_api_key", "key",
        "token", "access_token", "refresh_token", "id_token", "auth_token",
        "session_token", "hf_token", "jwt", "secret", "client_secret", "private_key",
        "password", "passwd", "pwd", "passphrase", "credential", "credentials",
        "sig", "signature", "sas", "auth", "authorization", "bearer", "code",
        "subscription_key", "ocp_apim_subscription_key", "x_amz_signature",
        "x_amz_credential", "x_amz_security_token",
    }
)  # fmt: skip
_ZERO_WIDTH = dict.fromkeys((0x200B, 0x200C, 0x200D, 0x2060, 0xFEFF))
_CREDENTIALS_IN_URL = (
    "base_url must not carry credentials (user:secret@host); add the key as a provider key"
)
_UNPARSEABLE = "base_url is not a valid URL (it needs a host and at most a numeric port)"
# URL userinfo anywhere in free text, greedy to the last "@" of a whitespace-free run.
_USERINFO_IN_TEXT = re.compile(r"(?<=://)\S*@")
# A query or fragment parameter in free text: (separator + name + "=", value).
_PARAM_IN_TEXT = re.compile(
    r"([?&;#][^=&;#?\s'\"]+=)"
    r"(\\?(['\"]).*?\\?\3[^&;#\s]*|[^&;#\s]*)"
)
_PARAM_SEPARATOR = re.compile(r"[&;]")


def _normalized_name(raw: str) -> str:
    name = unicodedata.normalize("NFKC", unquote_plus(raw)).translate(_ZERO_WIDTH)
    name = re.sub(r"[-+.\s]", "_", name.strip().lower())
    name = re.sub(r"(\[[^\[\]]*\])+$", "", name)
    return re.sub(r"_+", "_", name).strip("_")


def _name_problem(raw: str) -> str | None:
    name = _normalized_name(raw)
    if name in _CREDENTIAL_PARAMS:
        return (
            "base_url must not carry a credential in a query, fragment or path "
            f"parameter (parameter {name!r}); add the key as a provider key"
        )
    decoded = unquote_plus(raw)
    if not decoded.isascii() or any(
        c.isspace() or unicodedata.category(c).startswith("C") for c in decoded
    ):
        return (
            "base_url has a parameter whose decoded name is not plain ASCII; it cannot be "
            "checked for a credential, so it is refused"
        )
    return None


def _param_names(parts: SplitResult) -> list[str]:
    names = [
        p.partition("=")[0]
        for section in (parts.query, parts.fragment)
        for p in _PARAM_SEPARATOR.split(section)
        if p
    ]
    for segment in parts.path.split("/"):
        names.extend(p.partition("=")[0] for p in segment.split(";")[1:] if p)
    return names


def base_url_problem(url: str) -> str | None:
    """Why `url` cannot be a provider base URL, or None. Never repeats any part of the URL."""
    if any(c.isspace() or unicodedata.category(c).startswith("C") for c in url):
        return "base_url must not contain whitespace or control characters"
    try:
        parts = urlsplit(url)
        _ = parts.port  # raises ValueError for a non-numeric or out-of-range port
        host = parts.hostname
    except ValueError:
        return _UNPARSEABLE
    if parts.scheme not in ("http", "https"):
        return "base_url must start with http:// or https://"
    if "@" in parts.netloc:
        return _CREDENTIALS_IN_URL
    if not host:
        return "base_url has no host"
    for raw in _param_names(parts):
        problem = _name_problem(raw)
        if problem is not None:
            return problem
    return None


def scrub_url_secrets(text: str) -> str:
    """Every URL userinfo and every query or fragment parameter value in `text` masked."""
    text = _USERINFO_IN_TEXT.sub("***@", text)
    return _PARAM_IN_TEXT.sub(lambda m: m.group(1) + MASK, text)


def error_text(exc: BaseException) -> str:
    """`str(exc)` made safe to show. URL-parse errors are reduced to their kind, because their
    text can quote part of a key."""
    import httpx

    if isinstance(exc, httpx.InvalidURL):
        return "the URL could not be parsed (its text is withheld: it can quote part of a key)"
    return scrub_url_secrets(str(exc))


def _lower_hex(text: str) -> str:
    return re.sub(r"%[0-9A-F]{2}", lambda m: m.group(0).lower(), text)


def _forms(secret: str) -> set[str]:
    """`secret` verbatim, percent-encoded three ways (upper and lower hex), and base64 of
    `s`, `":" + s` and `s + ":"` (Basic auth with an empty user or password)."""
    percent = {quote(secret, safe=""), quote(secret), quote_plus(secret)}
    b64 = {
        base64.b64encode(raw.encode()).decode() for raw in (secret, ":" + secret, secret + ":")
    }
    return {secret, *percent, *map(_lower_hex, percent), *b64}


def scrub_values(text: str, secrets: Iterable[str]) -> str:
    """`text` with every occurrence of every non-empty secret, and its encoded forms, masked.
    Overlapping spans are merged first so no fragment of a secret survives."""
    spans: list[tuple[int, int]] = []
    for secret in {form for s in secrets if s for form in _forms(s)}:
        start = text.find(secret)
        while start != -1:
            spans.append((start, start + len(secret)))
            start = text.find(secret, start + 1)
    if not spans:
        return text
    spans.sort()
    merged = [spans[0]]
    for lo, hi in spans[1:]:
        if lo <= merged[-1][1]:
            merged[-1] = (merged[-1][0], max(merged[-1][1], hi))
        else:
            merged.append((lo, hi))
    out: list[str] = []
    pos = 0
    for lo, hi in merged:
        out.append(text[pos:lo])
        out.append(MASK)
        pos = hi
    out.append(text[pos:])
    return "".join(out)


def mask_key(value: str) -> str:
    """A hint for the UI: the last 4 characters, only when at least as many stay hidden."""
    return "••••" + value[-4:] if len(value) >= 8 else "••••"
```

```python
from __future__ import annotations

import asyncio
import os
import subprocess
from collections.abc import Awaitable, Callable, Sequence
from dataclasses import dataclass
from pathlib import Path

from mlops_agents.credentials.scrub import scrub_url_secrets, scrub_values

TOKEN_ENV = "MLOPS_AGENTS_GIT_TOKEN"
TokenProvider = Callable[[], Awaitable[str | None]]

# git asks GIT_ASKPASS for a username, then a password. GitHub accepts any username with a
# token as the password. The token travels only in the child's environment.
_ASKPASS_POSIX = (
    '#!/bin/sh\ncase "$1" in\n  Username*) echo x-access-token ;;\n'
    '  *) printf "%s\\n" "$MLOPS_AGENTS_GIT_TOKEN" ;;\nesac\n'
)
_ASKPASS_WINDOWS = (
    '@echo off\r\necho %~1| findstr /b /c:"Username" >nul\r\n'
    "if %errorlevel%==0 (echo x-access-token) else (echo %MLOPS_AGENTS_GIT_TOKEN%)\r\n"
)


async def no_token() -> str | None:
    return None


@dataclass(frozen=True)
class GitResult:
    code: int
    stdout: str
    stderr: str


class GitError(Exception):
    def __init__(self, message: str, *, code: int) -> None:
        super().__init__(message)
        self.code = code


class GitRunner:
    """Runs git without ever prompting: no credential helpers, no terminal prompts, no GCM
    windows. Tokens reach git only through GIT_ASKPASS; output is scrubbed of them."""

    def __init__(self, bin_dir: Path, token_provider: TokenProvider = no_token) -> None:
        self._bin_dir = bin_dir
        self._token_provider = token_provider

    def askpass_path(self) -> Path:
        windows = os.name == "nt"
        path = self._bin_dir / ("git-askpass.cmd" if windows else "git-askpass.sh")
        content = _ASKPASS_WINDOWS if windows else _ASKPASS_POSIX
        if not path.exists() or path.read_text(encoding="utf-8") != content:
            self._bin_dir.mkdir(parents=True, exist_ok=True)
            path.write_text(content, encoding="utf-8", newline="")
            if not windows:
                path.chmod(0o700)
        return path

    async def run(
        self,
        args: Sequence[str],
        cwd: Path | None = None,
        *,
        check: bool = True,
        timeout: float = 600,
        authenticated: bool = False,
    ) -> GitResult:
        token = await self._token_provider() if authenticated else None
        env = dict(os.environ)
        env.update(
            {
                "GIT_TERMINAL_PROMPT": "0",
                "GCM_INTERACTIVE": "never",
                "GIT_ASKPASS": str(self.askpass_path()),
            }
        )
        env.pop(TOKEN_ENV, None)
        if token:
            env[TOKEN_ENV] = token
        command = ["git", "-c", "credential.helper=", "-c", "core.quotepath=off", *args]
        try:
            completed = await asyncio.to_thread(
                subprocess.run, command, cwd=cwd, env=env, capture_output=True, timeout=timeout, check=False
            )
        except FileNotFoundError:
            raise GitError("git is not installed or not on PATH. Install git and restart the app.", code=-1) from None
        except subprocess.TimeoutExpired:
            raise GitError(f"git {args[0]} took longer than {int(timeout)} seconds and was stopped.", code=-1) from None
        hidden = [token] if token else []
        stdout = scrub_values(completed.stdout.decode("utf-8", "replace"), hidden)
        stderr = scrub_values(scrub_url_secrets(completed.stderr.decode("utf-8", "replace")), hidden)
        if check and completed.returncode != 0:
            reason = stderr.strip() or stdout.strip() or "no output"
            raise GitError(f"git {args[0]} failed: {reason}", code=completed.returncode)
        return GitResult(completed.returncode, stdout, stderr)
```
